function loginStudent(){

/* إضافة طالب */

function addStudent(){

let name=document.getElementById("studentName").value.trim();

let age=document.getElementById("studentAge").value.trim();

let address=document.getElementById("studentAddress").value.trim();

let grade=document.getElementById("studentGrade").value;

let level=document.getElementById("studentLevel").value;

let password=document.getElementById("studentPassword").value.trim();

let result=document.getElementById("studentResult").value.trim();

let subjectsText=document.getElementById("studentSubjects").value.trim();

let imageInput=document.getElementById("studentImage");

if(password.length!==6 || isNaN(password)){

alert("كلمة السر يجب أن تكون 6 أرقام");

return;

}

let subjects=[];

subjectsText.split('\n').forEach(line=>{

let parts=line.split(':');

if(parts.length===2){

subjects.push({

name:parts[0],

grade:parts[1]

});

}

});

let reader=new FileReader();

reader.onload=function(){

let students=JSON.parse(localStorage.getItem("students")) || [];

students.push({

id:Date.now(),

name,

age,

address,

grade,

level,

password,

result,

subjects,

image:reader.result,

feesPaid:false,

absent:0

});

localStorage.setItem("students",

JSON.stringify(students));

alert("✅ تم إضافة الطالب");

showAdminPanel();

}

if(imageInput.files[0]){

reader.readAsDataURL(imageInput.files[0]);

}else{

reader.onload();

}

}

/* تعديل */

function editStudent(id){

let students=JSON.parse(localStorage.getItem("students"));

let student=students.find(s=>s.id===id);

let newName=prompt("الاسم",student.name);

if(newName===null) return;

student.name=newName;

student.result=prompt("النتيجة",student.result);

student.grade=prompt("الصف",student.grade);

student.level=prompt("المستوى",student.level);

localStorage.setItem("students",

JSON.stringify(students));

alert("✅ تم التعديل");

showAdminPanel();

}

/* حذف */

function deleteStudent(id){

if(!confirm("حذف الطالب؟")) return;

let students=JSON.parse(localStorage.getItem("students"));

students=students.filter(s=>s.id!==id);

localStorage.setItem("students",

JSON.stringify(students));

showAdminPanel();

}

/* دفع */

function toggleFees(id){

let students=JSON.parse(localStorage.getItem("students"));

students=students.map(s=>{

if(s.id===id){

s.feesPaid=!s.feesPaid;

}

return s;

});

localStorage.setItem("students",

JSON.stringify(students));

showAdminPanel();

}

/* غياب */

function addAbsence(id){

let students=JSON.parse(localStorage.getItem("students"));

students=students.map(s=>{

if(s.id===id){

s.absent=(s.absent || 0)+1;

}

return s;

});

localStorage.setItem("students",

JSON.stringify(students));

showAdminPanel();

}

/* البحث */

function searchStudent(){

let input=document.getElementById("searchInput")

.value.toLowerCase();

let table=document.getElementById("studentsTable");

let tr=table.getElementsByTagName("tr");

for(let i=1;i<tr.length;i++){

let td=tr[i].getElementsByTagName("td")[0];

if(td){

let txt=td.textContent || td.innerText;

tr[i].style.display=

txt.toLowerCase().includes(input)

? ""

: "none";

}

}

}

const schoolDays=[

"السبت",

"الأحد",

"الإثنين",

"الثلاثاء",

"الأربعاء"

];

function loadScheduleInputs(){

let html="";

schoolDays.forEach(day=>{

html+=`

<tr>

<td>${day}</td>

<td><input id="${day}1"></td>

<td><input id="${day}2"></td>

<td><input id="${day}3"></td>

<td><input id="${day}4"></td>

<td><input id="${day}5"></td>

<td><input id="${day}6"></td>

</tr>

`;

});

setTimeout(()=>{

document.getElementById("scheduleInputs").innerHTML=html;

},100);

}

function saveSchedule(){

let grade=document.getElementById("tableGrade").value;

let schedules=

JSON.parse(localStorage.getItem("schedules")) || {};

schedules[grade]={};

schoolDays.forEach(day=>{

schedules[grade][day]=[];

for(let i=1;i<=6;i++){

schedules[grade][day].push(

document.getElementById(`${day}${i}`).value

);

}

});

localStorage.setItem(

"schedules",

JSON.stringify(schedules)

);

alert("✅ تم حفظ الجدول الدراسي");

}
